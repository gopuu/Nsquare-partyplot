CREATE TABLE AccountMaster
(
    Id BIGINT IDENTITY(1,1) PRIMARY KEY,

    AccountName NVARCHAR(150) NOT NULL,          -- Display Name
    AccountType NVARCHAR(50) NOT NULL,           -- Cash, Bank, UPI, CreditCard, Wallet

    OpeningBalance DECIMAL(18,2) DEFAULT 0,
    CurrentBalance DECIMAL(18,2) DEFAULT 0,

    -- Bank Details (nullable for Cash/Wallet/UPI accounts)
    BankName NVARCHAR(150) NULL,
    BranchName NVARCHAR(150) NULL,
    IFSC NVARCHAR(20) NULL,
    AccountNumber NVARCHAR(50) NULL,
    AccountHolderName NVARCHAR(150) NULL,

    -- UPI field for UPI accounts
    UPI_Id NVARCHAR(100) NULL,

    Description NVARCHAR(300) NULL,

    -- Flags
    IsDefaultAccount BIT DEFAULT 0,  -- Useful for the system's primary account

    -- Audit Columns
    is_deleted BIT DEFAULT 0,
    created_at DATETIME DEFAULT dbo.GETDATEIST(),
    created_by BIGINT NULL,
    updated_at DATETIME NULL,
    updated_by BIGINT NULL,
    deleted_at DATETIME NULL,
    deleted_by BIGINT NULL
);

CREATE TABLE AccountBalanceAudit
(
    Id BIGINT IDENTITY(1,1) PRIMARY KEY,

    AccountId BIGINT NOT NULL,
    PreviousBalance DECIMAL(18,2) NOT NULL,
    AmountChanged DECIMAL(18,2) NOT NULL,  -- +Credit / -Debit
    NewBalance DECIMAL(18,2) NOT NULL,

    ActionType NVARCHAR(100) NOT NULL,      -- Income, Expense, FundTransfer-Debit, etc.
    ReferenceId BIGINT NULL,                -- Transaction Id (Income, Expense, FundTransfer)

    created_at DATETIME DEFAULT dbo.GETDATEIST(),
    created_by BIGINT NULL
);
